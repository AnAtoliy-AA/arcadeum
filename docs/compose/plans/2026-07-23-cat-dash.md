# Cat Dash Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use compose:subagent (recommended) or compose:execute to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a multiplayer cat racing dice game with host-selectable tracks and themes, supporting 2-6 players with unique cat characters and special abilities.

**Architecture:** Two-half architecture following existing game patterns: backend engine for pure game logic, frontend widget for UI. Track types and themes are configurable per game room. Cat characters have unique abilities that affect gameplay.

**Tech Stack:** TypeScript, NestJS backend, React/Next.js frontend, Zustand state management, Tamagui UI components, Socket.IO real-time communication.

```typescript
// apps/be/src/games/engines/cat-dash/cat-dash.utils.ts
import { TrackSpace, SpaceEffect } from './cat-dash.types';
import { TrackType, TRACK_LENGTH } from './cat-dash.constants';

export function generateTrack(trackType: TrackType): TrackSpace[] {
  const track: TrackSpace[] = [];

  for (let i = 0; i <= TRACK_LENGTH; i++) {
    const space: TrackSpace = { id: i, type: 'normal' };

    if (i === 0) {
      space.type = 'normal';
    } else if (i === TRACK_LENGTH) {
      space.type = 'normal';
    } else if (i % 5 === 0) {
      space.type = 'obstacle';
      space.effect = { type: 'skip_turn' };
    } else if (i % 3 === 0) {
      space.type = 'bonus';
      space.effect = { type: 'extra_roll' };
    }

    track.push(space);
  }

  return track;
}

export function calculateMovement(
  baseRoll: number,
  hasBonus: boolean,
  abilityModifier: number,
): number {
  let movement = baseRoll;
  if (hasBonus) movement += 1;
  movement += abilityModifier;
  return Math.max(1, movement);
}

export function checkWinCondition(position: number): boolean {
  return position >= TRACK_LENGTH;
}

export function getAvailableAbilities(
  catId: string,
  abilitiesUsed: string[],
): string[] {
  return ['ability_1', 'ability_2'].filter(
    (ability) => !abilitiesUsed.includes(ability),
  );
}

export function applySpaceEffect(effect: SpaceEffect): {
  skipTurn: boolean;
  extraRoll: boolean;
  powerRecharge: boolean;
} {
  return {
    skipTurn: effect.type === 'skip_turn',
    extraRoll: effect.type === 'extra_roll',
    powerRecharge: effect.type === 'power_recharge',
  };
}
```

```typescript
// apps/be/src/games/engines/cat-dash/cat-dash.validators.ts
import { State, GameAction } from './cat-dash.types';

export function validateAction(
  state: State,
  action: GameAction,
): { ok: true } | { ok: false; error: string } {
  if (state.gameOver) {
    return { ok: false, error: 'Game is over' };
  }

  const currentPlayer = state.players[state.currentPlayerIndex];
  if (!currentPlayer || currentPlayer.id !== action.playerId) {
    return { ok: false, error: 'Not your turn' };
  }

  switch (action.type) {
    case 'ROLL_DICE':
      return { ok: true };

    case 'MOVE':
      if (action.spaces < 1 || action.spaces > 6) {
        return { ok: false, error: 'Invalid movement' };
      }
      return { ok: true };

    case 'USE_ABILITY':
      if (currentPlayer.abilitiesUsed.includes(action.abilityId)) {
        return { ok: false, error: 'Ability already used' };
      }
      if (currentPlayer.powerTokens <= 0) {
        return { ok: false, error: 'No power tokens' };
      }
      return { ok: true };

    case 'CHOOSE_PATH':
      if (action.pathIndex < 0 || action.pathIndex > 2) {
        return { ok: false, error: 'Invalid path' };
      }
      return { ok: true };

    default:
      return { ok: false, error: 'Unknown action' };
  }
}
```

```typescript
// apps/be/src/games/engines/cat-dash/cat-dash.engine.ts
import { BaseGameEngine } from '../base/base-game.engine';
import { State, Options, Player, GameAction } from './cat-dash.types';
import {
  MIN_PLAYERS,
  MAX_PLAYERS,
  CAT_IDS,
  DEFAULT_OPTIONS,
  POWER_TOKENS_PER_GAME,
} from './cat-dash.constants';
import {
  generateTrack,
  calculateMovement,
  checkWinCondition,
  applySpaceEffect,
} from './cat-dash.utils';
import { validateAction } from './cat-dash.validators';
import { createLogEntry } from '../base/game-log';

export class CatDashEngine extends BaseGameEngine<State> {
  getMetadata() {
    return {
      id: 'cat-dash_v1',
      name: 'Cat Dash',
      minPlayers: MIN_PLAYERS,
      maxPlayers: MAX_PLAYERS,
      variants: ['linear', 'circular', 'multiple'],
      defaultOptions: DEFAULT_OPTIONS,
    };
  }

  initializeState(playerIds: string[], options: Partial<Options> = {}): State {
    const trackType = options.trackType || DEFAULT_OPTIONS.trackType;
    const theme = options.theme || DEFAULT_OPTIONS.theme;

    const players: Player[] = playerIds.map((id, index) => ({
      id,
      catId: CAT_IDS[index % CAT_IDS.length],
      position: 0,
      powerTokens: POWER_TOKENS_PER_GAME,
      abilitiesUsed: [],
      isReady: true,
      hasBonus: false,
    }));

    return {
      trackType,
      theme,
      players,
      currentPlayerIndex: 0,
      turnNumber: 1,
      track: generateTrack(trackType),
      gameOver: false,
      logs: [],
    };
  }

  validateAction(
    state: State,
    action: GameAction,
  ): { ok: true } | { ok: false; error: string } {
    return validateAction(state, action);
  }

  executeAction(state: State, action: GameAction): State {
    const validation = this.validateAction(state, action);
    if (!validation.ok) {
      throw new Error(validation.error);
    }

    const newState = { ...state };
    const playerIndex = newState.players.findIndex(
      (p) => p.id === action.playerId,
    );
    const player = { ...newState.players[playerIndex] };

    switch (action.type) {
      case 'ROLL_DICE': {
        const roll = Math.floor(Math.random() * 6) + 1;
        const movement = calculateMovement(roll, player.hasBonus, 0);

        newState.logs.push(
          createLogEntry(
            action.playerId,
            'roll',
            `Rolled ${roll}, moving ${movement} spaces`,
          ),
        );

        player.position = Math.min(
          player.position + movement,
          newState.track.length - 1,
        );
        player.hasBonus = false;

        const space = newState.track[player.position];
        if (space.effect) {
          const effects = applySpaceEffect(space.effect);
          if (effects.skipTurn) {
            newState.logs.push(
              createLogEntry(
                action.playerId,
                'obstacle',
                'Hit an obstacle, skipping next turn',
              ),
            );
          }
          if (effects.extraRoll) {
            newState.logs.push(
              createLogEntry(
                action.playerId,
                'bonus',
                'Found a bonus, rolling again',
              ),
            );
          }
        }

        if (checkWinCondition(player.position)) {
          newState.winner = action.playerId;
          newState.gameOver = true;
          newState.logs.push(
            createLogEntry(action.playerId, 'win', 'Crossed the finish line!'),
          );
        }

        break;
      }

      case 'MOVE': {
        player.position = Math.min(
          player.position + action.spaces,
          newState.track.length - 1,
        );

        newState.logs.push(
          createLogEntry(
            action.playerId,
            'move',
            `Moved ${action.spaces} spaces`,
          ),
        );

        if (checkWinCondition(player.position)) {
          newState.winner = action.playerId;
          newState.gameOver = true;
        }

        break;
      }

      case 'USE_ABILITY': {
        player.abilitiesUsed.push(action.abilityId);
        player.powerTokens -= 1;

        newState.logs.push(
          createLogEntry(
            action.playerId,
            'ability',
            `Used ability ${action.abilityId}`,
          ),
        );

        break;
      }

      case 'CHOOSE_PATH': {
        newState.logs.push(
          createLogEntry(
            action.playerId,
            'path',
            `Chose path ${action.pathIndex}`,
          ),
        );

        break;
      }
    }

    newState.players[playerIndex] = player;

    if (!newState.gameOver) {
      newState.currentPlayerIndex =
        (newState.currentPlayerIndex + 1) % newState.players.length;
      if (newState.currentPlayerIndex === 0) {
        newState.turnNumber += 1;
      }
    }

    return newState;
  }

  isGameOver(state: State): boolean {
    return state.gameOver;
  }

  getWinners(state: State): string[] {
    return state.winner ? [state.winner] : [];
  }

  sanitizeStateForPlayer(state: State, playerId: string): State {
    return {
      ...state,
      players: state.players.map((p) => ({
        ...p,
        abilitiesUsed: p.id === playerId ? p.abilitiesUsed : [],
      })),
    };
  }

  getAvailableActions(state: State, playerId: string): GameAction[] {
    const currentPlayer = state.players[state.currentPlayerIndex];
    if (!currentPlayer || currentPlayer.id !== playerId || state.gameOver) {
      return [];
    }

    const actions: GameAction[] = [{ type: 'ROLL_DICE', playerId }];

    if (currentPlayer.powerTokens > 0) {
      const availableAbilities = ['ability_1', 'ability_2'].filter(
        (ability) => !currentPlayer.abilitiesUsed.includes(ability),
      );

      availableAbilities.forEach((abilityId) => {
        actions.push({ type: 'USE_ABILITY', playerId, abilityId });
      });
    }

    return actions;
  }
}
```

```typescript
// apps/be/src/games/cat-dash/cat-dash.service.ts
import { Injectable } from '@nestjs/common';
import { CatDashEngine } from '../engines/cat-dash/cat-dash.engine';
import { State, GameAction } from '../engines/cat-dash/cat-dash.types';
import { GameRoomsService } from './rooms/game-rooms.service';
import { GameSessionsService } from './game-sessions.service';
import { GamesRealtimeService } from './games-realtime.service';

@Injectable()
export class CatDashService {
  constructor(
    private readonly engine: CatDashEngine,
    private readonly roomsService: GameRoomsService,
    private readonly sessionsService: GameSessionsService,
    private readonly realtimeService: GamesRealtimeService,
  ) {}

  async startSession(
    roomId: string,
    playerIds: string[],
    options: any,
  ): Promise<State> {
    const state = this.engine.initializeState(playerIds, options);
    await this.sessionsService.createSession(roomId, state);
    return state;
  }

  async handleAction(roomId: string, action: GameAction): Promise<State> {
    const session = await this.sessionsService.getSession(roomId);
    if (!session) {
      throw new Error('Session not found');
    }

    const validation = this.engine.validateAction(session.state, action);
    if (!validation.ok) {
      throw new Error(validation.error);
    }

    const newState = this.engine.executeAction(session.state, action);
    await this.sessionsService.updateSession(roomId, newState);

    await this.realtimeService.broadcastToRoom(roomId, 'gameUpdate', newState);

    return newState;
  }

  async getState(roomId: string): Promise<State | null> {
    const session = await this.sessionsService.getSession(roomId);
    return session?.state || null;
  }
}
```

```typescript
// apps/be/src/games/cat-dash/cat-dash-bot.service.ts
import { Injectable } from '@nestjs/common';
import { CatDashEngine } from '../engines/cat-dash/cat-dash.engine';
import { State, GameAction } from '../engines/cat-dash/cat-dash.types';

@Injectable()
export class CatDashBotService {
  constructor(private readonly engine: CatDashEngine) {}

  async pickMove(state: State, botId: string): Promise<GameAction> {
    const availableActions = this.engine.getAvailableActions(state, botId);

    if (availableActions.length === 0) {
      return { type: 'ROLL_DICE', playerId: botId };
    }

    const rollAction = availableActions.find((a) => a.type === 'ROLL_DICE');
    if (rollAction) {
      return rollAction;
    }

    return availableActions[0];
  }
}
```

```typescript
import {
  WebSocketGateway,
  SubscribeMessage,
  ConnectedSocket,
  MessageBody,
} from '@nestjs/websockets';
import { Socket } from 'socket.io';
import { CatDashService } from './cat-dash/cat-dash.service';
import { GameAction } from './engines/cat-dash/cat-dash.types';

@WebSocketGateway({ cors: true })
export class CatDashGateway {
  constructor(private readonly service: CatDashService) {}

  @SubscribeMessage('catDash.session.start')
  async handleStart(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string; playerIds: string[]; options: any },
  ) {
    try {
      const state = await this.service.startSession(
        data.roomId,
        data.playerIds,
        data.options,
      );
      client.emit('catDash.session.started', { state });
      return { success: true, state };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  @SubscribeMessage('catDash.session.action')
  async handleAction(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string; action: GameAction },
  ) {
    try {
      const state = await this.service.handleAction(data.roomId, data.action);
      return { success: true, state };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }

  @SubscribeMessage('catDash.session.state')
  async handleGetState(
    @ConnectedSocket() client: Socket,
    @MessageBody() data: { roomId: string },
  ) {
    try {
      const state = await this.service.getState(data.roomId);
      return { success: true, state };
    } catch (error) {
      return { success: false, error: error.message };
    }
  }
}
```

```typescript
import { useMemo } from 'react';
import { CatDashState, CatDashPlayer } from '../types';

interface CatDashSnapshot {
  state: CatDashState | null;
  currentPlayer: CatDashPlayer | null;
  myPlayer: CatDashPlayer | null;
  myTurn: boolean;
  isGameOver: boolean;
  winner: CatDashPlayer | undefined;
  trackSpaces: number;
}

export function useStateSelector(
  state: CatDashState | null,
  currentUserId: string,
): CatDashSnapshot {
  return useMemo(() => {
    if (!state) {
      return {
        state: null,
        currentPlayer: null,
        myPlayer: null,
        myTurn: false,
        isGameOver: false,
        winner: undefined,
        trackSpaces: 0,
      };
    }

    const currentPlayer = state.players[state.currentPlayerIndex];
    const myPlayer = state.players.find((p) => p.id === currentUserId);
    const myTurn = currentPlayer?.id === currentUserId;
    const winner = state.players.find((p) => p.id === state.winner);

    return {
      state,
      currentPlayer,
      myPlayer,
      myTurn,
      isGameOver: state.gameOver,
      winner,
      trackSpaces: state.track.length,
    };
  }, [state, currentUserId]);
}
```

```typescript
import { useCallback } from 'react';
import { gameSocket } from '@/shared/lib/socket';
import { GameAction } from '../types';

export function useActions(roomId: string) {
  const rollDice = useCallback(
    (playerId: string) => {
      const action: GameAction = { type: 'ROLL_DICE', playerId };
      gameSocket.emit('catDash.session.action', { roomId, action });
    },
    [roomId],
  );

  const move = useCallback(
    (playerId: string, spaces: number) => {
      const action: GameAction = { type: 'MOVE', playerId, spaces };
      gameSocket.emit('catDash.session.action', { roomId, action });
    },
    [roomId],
  );

  const useAbility = useCallback(
    (playerId: string, abilityId: string) => {
      const action: GameAction = { type: 'USE_ABILITY', playerId, abilityId };
      gameSocket.emit('catDash.session.action', { roomId, action });
    },
    [roomId],
  );

  const choosePath = useCallback(
    (playerId: string, pathIndex: number) => {
      const action: GameAction = { type: 'CHOOSE_PATH', playerId, pathIndex };
      gameSocket.emit('catDash.session.action', { roomId, action });
    },
    [roomId],
  );

  return {
    rollDice,
    move,
    useAbility,
    choosePath,
  };
}
```

```typescript
// apps/be/src/games/engines/cat-dash/cat-dash.engine.spec.ts
import { CatDashEngine } from './cat-dash.engine';

describe('CatDashEngine', () => {
  let engine: CatDashEngine;

  beforeEach(() => {
    engine = new CatDashEngine();
  });

  it('should initialize state correctly', () => {
    const state = engine.initializeState(['player1', 'player2']);

    expect(state.players).toHaveLength(2);
    expect(state.trackType).toBe('linear');
    expect(state.gameOver).toBe(false);
    expect(state.turnNumber).toBe(1);
  });

  it('should validate roll dice action', () => {
    const state = engine.initializeState(['player1', 'player2']);
    const action = { type: 'ROLL_DICE' as const, playerId: 'player1' };

    const result = engine.validateAction(state, action);
    expect(result.ok).toBe(true);
  });

  it('should reject action from wrong player', () => {
    const state = engine.initializeState(['player1', 'player2']);
    const action = { type: 'ROLL_DICE' as const, playerId: 'player2' };

    const result = engine.validateAction(state, action);
    expect(result.ok).toBe(false);
  });

  it('should execute roll dice action', () => {
    const state = engine.initializeState(['player1', 'player2']);
    const action = { type: 'ROLL_DICE' as const, playerId: 'player1' };

    const newState = engine.executeAction(state, action);
    expect(newState.players[0].position).toBeGreaterThan(0);
    expect(newState.currentPlayerIndex).toBe(1);
  });

  it('should detect win condition', () => {
    const state = engine.initializeState(['player1', 'player2']);
    state.players[0].position = 19;
    const action = { type: 'ROLL_DICE' as const, playerId: 'player1' };

    const newState = engine.executeAction(state, action);
    expect(newState.gameOver).toBe(true);
    expect(newState.winner).toBe('player1');
  });
});
```
